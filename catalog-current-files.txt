"use client";

import { useMemo, useState } from "react";

import {
  cakes,
  type Cake,
  type CakeCategory,
} from "@/lib/cakes";

import CakeCategories from "@/app/components/cakes/CakeCategories";
import CakeSearch from "@/app/components/cakes/CakeSearch";
import CakeSort, { type CakeSortType } from "@/app/components/cakes/CakeSort";
import CakesGrid from "@/app/components/cakes/CakesGrid";

export default function CakesPage() {
  const [category, setCategory] =
    useState<CakeCategory>("all");

  const [search, setSearch] =
    useState("");

  const [sort, setSort] =
    useState<CakeSortType>("popular");

  const filteredCakes = useMemo(() => {
    let data: Cake[] = [...cakes];

    if (category !== "all") {
      data = data.filter(
        (cake) => cake.category === category
      );
    }

    if (search.trim()) {
      const q = search.toLowerCase();

      data = data.filter(
        (cake) =>
          cake.name.toLowerCase().includes(q) ||
          cake.description
            .toLowerCase()
            .includes(q)
      );
    }

    switch (sort) {
      case "popular":
        data.sort(
          (a, b) =>
            Number(b.popular) -
            Number(a.popular)
        );
        break;

      case "new":
        data.sort(
          (a, b) =>
            Number(b.isNew) -
            Number(a.isNew)
        );
        break;

      case "priceAsc":
        data.sort(
          (a, b) => a.price - b.price
        );
        break;

      case "priceDesc":
        data.sort(
          (a, b) => b.price - a.price
        );
        break;
    }

    return data;
  }, [
    category,
    search,
    sort,
  ]);

  return (
    <main className="min-h-screen bg-[#faf8f6]">

      <section className="mx-auto max-w-7xl px-6 py-20">

        <div className="mb-14 text-center">

          <h1 className="text-5xl font-bold">
            Р“РѕС‚РѕРІС‹Рµ С‚РѕСЂС‚С‹
          </h1>

          <p className="mx-auto mt-6 max-w-2xl text-lg text-black/60">
            Р’С‹Р±РµСЂРёС‚Рµ РіРѕС‚РѕРІС‹Р№ РґРёР·Р°Р№РЅ РёР»Рё
            РёР·РјРµРЅРёС‚Рµ РµРіРѕ РІ РЅР°С€РµРј
            РєРѕРЅСЃС‚СЂСѓРєС‚РѕСЂРµ.
          </p>

        </div>

        <div className="space-y-8">

          <CakeCategories
            value={category}
            onChange={setCategory}
          />

          <div className="flex flex-col gap-4 lg:flex-row">

            <div className="flex-1">
              <CakeSearch
                value={search}
                onChange={setSearch}
              />
            </div>

            <CakeSort
              value={sort}
              onChange={setSort}
            />

          </div>

          <div className="flex items-center justify-between">

  value,
  onChange,
}: Props) {
  return (
    <div className="flex flex-wrap gap-3">
      {cakeCategories.map((category) => {
        const active = value === category.value;

        return (
          <button
            key={category.value}
